import argparse

# Automatically locates the script folder and pins the profile file there
SCRIPT_DIR = Path(__file__).resolve().parent
PROFILE_FILE = SCRIPT_DIR / "audit_key_profile.json"

# Operational structural keys skipped during active cleanup prompts
BYPASS_KEYS = (
    "block_type", 
    "Audit_Source_Type", 
    "Original_File", 
    "Block_Scope", 
    "Condition_Context", 
    "Condition_Block_Context", 
    "Report_Context"
)


def load_profile() -> dict:
    """Loads existing key classifications from the script's JSON profile file."""
    if PROFILE_FILE.exists():
        try:
            with PROFILE_FILE.open("r", encoding="utf-8") as f:
                return json.load(f)
        except Exception as e:
            print(f"Warning: Failed to load profile ({e}). Starting fresh.")
    return {}


def save_profile(profile_data: dict):
    """Saves updated key classifications back to the JSON profile file."""
    try:
        with PROFILE_FILE.open("w", encoding="utf-8") as f:
            json.dump(profile_data, f, indent=4, sort_keys=True)
        print(f"\n[Profile Sync] Key map configurations saved to '{PROFILE_FILE}'")
    except Exception as e:
        print(f"Error: Could not save profile file ({e})")


def detect_audit_type(audit_text: str, filename: str = "") -> str:
    """
    Groups audit files with absolute precision by matching explicit naming signatures 
    in the filename first, falling back strictly to the internal text content.
    """
    fn = filename.upper() if filename else ""
    print(f" -> Inspecting file pattern: {filename if filename else 'Raw Text'}")
    
    # 1. STRICT FILE NAME EVALUATION (Highest Priority)
    if fn:
        if "AMAZON" in fn or "AWS" in fn:
            return "AWS_Foundations"
        if "AZURE" in fn:
            return "Azure_Foundations"
        if "NX-OS" in fn or "NXOS" in fn:
            return "Cisco_NXOS"
        if "IOS" in fn:
            return "Cisco_IOS"
        if "SQL" in fn:
            return "MS_SQL_Server"
        if "WINDOWS_11" in fn or "WINDOWS11" in fn or "WIN_11" in fn:
            return "Windows_11_Enterprise"
        if "WINDOWS_SERVER" in fn or ("WINDOWS" in fn and "SERVER" in fn):
            return "Windows_Server"

    # 2. INTERNAL TEXT CONTENT FALLBACK (If filename is missing or generic)
    title_m = re.search(r"^\s*title\s*:\s*\"?([^\"]+)\"?", audit_text, flags=re.IGNORECASE | re.MULTILINE)
    title_text = title_m.group(1).strip().upper() if title_m else ""

    if "AWS" in title_text or "AMAZON" in title_text:
        return "AWS_Foundations"
    if "AZURE" in title_text:
        return "Azure_Foundations"
    if "NX-OS" in title_text or "NXOS" in title_text:
        return "Cisco_NXOS"
    if "IOS" in title_text or "CISCO" in title_text:
        return "Cisco_IOS"
    if "SQL" in title_text:
        return "MS_SQL_Server"
    if "WINDOWS 11" in title_text or "WIN 11" in title_text:
        return "Windows_11_Enterprise"
    if "WINDOWS SERVER" in title_text:
        return "Windows_Server"
        
    check_type_m = re.search(r"<check_type\s*:\s*\"?([^\">]+)\"?", audit_text, flags=re.IGNORECASE)
    if check_type_m:
        plugin = check_type_m.group(1).strip().upper()
        if "WINDOWS" in plugin:
            return "Windows_Server"
        return plugin.replace(" ", "_")

    return "Unknown_Platform"


def parse_custom_items(audit_text: str):
    """
    Parses audit content using an accurate state stack to map raw <condition...> 
    and <report...> tag contexts down into explicit row metadata columns.
    """
    blocks = []
    lines = audit_text.splitlines()
    
    in_block = False
    active_conditions = []
    active_reports = []
    
    current_block = {}
    current_block_type = None
    current_key = None
    current_value_lines = []
    in_multiline_value = False

    def flush_key():
        nonlocal current_key, current_value_lines, current_block, in_multiline_value
        if current_key is None:
            return
        value = "\n".join(current_value_lines).rstrip()
        if value.startswith('"') and value.endswith('"') and len(value) >= 2:
            value = value[1:-1]
        current_block[current_key] = value
        current_key = None
        current_value_lines = []
        in_multiline_value = False

    # FIX: Clean regex anchors to correctly process standard .audit data configurations
    key_pattern = re.compile(r"^\s*([a-zA-Z0-9_]+)\s*:\s*(.*)$")
    start_pattern = re.compile(r"^\s*<(custom_item|item)>\s*$")
    end_pattern = re.compile(r"^\s*</(custom_item|item)>\s*$")
    
    condition_start = re.compile(r"^\s*(<condition\s*[^>]*>)\s*$", re.IGNORECASE)
    condition_end = re.compile(r"^\s*</condition>\s*$", re.IGNORECASE)
    
    report_start = re.compile(r"^\s*(<report\s*[^>]*>)\s*$", re.IGNORECASE)
    report_end = re.compile(r"^\s*</report>\s*$", re.IGNORECASE)


    for line in lines:
        c_start_match = condition_start.match(line)
        if c_start_match:
            active_conditions.append(c_start_match.group(1).strip())
            continue
        if condition_end.match(line):
            if active_conditions:
                active_conditions.pop()
            continue

        r_start_match = report_start.match(line)
        if r_start_match:
            active_reports.append(r_start_match.group(1).strip())
            continue
        if report_end.match(line):
            if active_reports:
                active_reports.pop()
            continue

        if not in_block:
            start_match = start_pattern.match(line)
            if start_match:
                in_block = True
                current_block = {}
                current_block_type = start_match.group(1)
                current_key = None
                current_value_lines = []
                in_multiline_value = False
            continue

        if end_pattern.match(line):
            flush_key()
            current_block["block_type"] = current_block_type
            current_block["Block_Scope"] = "Conditional Check" if active_conditions else "Standard Rule"
            
            current_block["Condition_Context"] = active_conditions[-1] if active_conditions else "N/A"
            current_block["Report_Context"] = active_reports[-1] if active_reports else "N/A"
            
            blocks.append(current_block)
            in_block = False
            current_block_type = None
            continue

        if current_key is not None and in_multiline_value:
            current_value_lines.append(line)
            if line.rstrip().endswith('"'):
                flush_key()
            continue

        match = key_pattern.match(line)
        if match:
            flush_key()
            current_key = match.group(1)
            val_part = match.group(2).lstrip()

            if val_part.startswith('"') and not val_part.rstrip().endswith('"'):
                in_multiline_value = True
                current_value_lines.append(val_part)
            elif (
                val_part.startswith('"')
                and val_part.rstrip().endswith('"')
                and len(val_part.strip()) >= 2
            ):
                current_value_lines.append(val_part)
                flush_key()
            else:
                current_value_lines.append(val_part)
                flush_key()

    return blocks


def parse_variables(audit_text: str):
    """
    Strips away comment tags (#) from the header file metadata to cleanly 
    isolate and resolve dynamic default validation parameters.
    """
    vars_map = {}
    
    # Clean the audit text block by stripping away leading '#' characters
    cleaned_lines = []
    for line in audit_text.splitlines():
        stripped = line.lstrip()
        if stripped.startswith("#"):
            cleaned_lines.append(stripped[1:])
        else:
            cleaned_lines.append(line)
            
    uncommented_text = "\n".join(cleaned_lines)
    
    # Locate all variable definitions
    variable_blocks = re.findall(
        r"<variable\s*>(.*?)</variable\s*>", uncommented_text, flags=re.IGNORECASE | re.DOTALL
    )

    for vb in variable_blocks:
        name_m = re.search(r"<name>\s*([A-Za-z0-9_]+)\s*</name>", vb, flags=re.IGNORECASE)
        if not name_m:
            # Fallback to standard key-value check format if tags are missing tags layout
            name_m = re.search(r"name\s*:\s*\"?([A-Za-z0-9_]+)\"?", vb, flags=re.IGNORECASE)
            
        if not name_m:
            continue

        name = name_m.group(1).strip()
        value_m = re.search(r"<value>\s*(.*?)\s*</value>", vb, flags=re.IGNORECASE | re.DOTALL)
        default_m = re.search(r"<default>\s*(.*?)\s*</default>", vb, flags=re.IGNORECASE | re.DOTALL)
        resolved = None

        if value_m:
            resolved = value_m.group(1).strip()
        elif default_m:
            resolved = default_m.group(1).strip()

        if resolved is not None:
            vars_map[name] = resolved

    return vars_map


def process_tokens(blocks, vars_map):
    """Replaces placeholder tokens like @PASSWORD_HISTORY@ with their verified parameter values."""
    token_pat = re.compile(r"@([A-Za-z0-9_]+)@")
    for b in blocks:
        for k, v in list(b.items()):
            if isinstance(v, str) and "@" in v:
                def _repl(m):
                    var = m.group(1)
                    return vars_map.get(var, m.group(0))
                b[k] = token_pat.sub(_repl, v)


def validate_and_classify_keys(all_files_data):
    """Loads cache profile and prompts ONLY for unclassified new keys, grouped by audit type."""
    profile = load_profile()
    profile_has_changed = False

    type_to_keys = {}
    for item in all_files_data:
        audit_type = item["audit_type"]
        if audit_type not in type_to_keys:
            type_to_keys[audit_type] = set()
        for b in item["blocks"]:
            type_to_keys[audit_type].update(b.keys())

    print("\n=======================================================")
    print("           KEY VALIDATION & CLASSIFICATION GATE         ")
    print("=======================================================")
    print("Options per key:\n  [a] = Active\n  [i] = Information\n  [n/Enter] = Drop key\n")

    for audit_type, keys in sorted(type_to_keys.items()):
        for bk in BYPASS_KEYS:
            keys.discard(bk)
            
        if not keys:
            continue

        if audit_type not in profile:
            profile[audit_type] = {}

        new_keys = [k for k in keys if k not in profile[audit_type]]

        print(f"\n--> AUDIT GROUP TYPE: [{audit_type.upper()}]")
        if not new_keys:
            print("    All keys found for this type match your profile. No new prompts.")
            continue

        print(f"    Found {len(new_keys)} NEW unclassified keys to review for this group.")
        print("-" * 65)

        for key in sorted(new_keys):
            while True:
                response = (
                    input(f"    [{audit_type}] -> Key '{key}' classification? [a/i/n]: ")
                    .strip()
                    .lower()
                )
                if response == "a":
                    profile[audit_type][key] = "Active"
                    profile_has_changed = True
                    break
                elif response == "i":
                    profile[audit_type][key] = "Information"
                    profile_has_changed = True
                    break
                elif response in ("n", ""):
                    profile[audit_type][key] = "Dropped"
                    profile_has_changed = True
                    print(f"    -> Marked '{key}' as Dropped")
                    break
                else:
                    print("    Invalid selection. Please choose 'a', 'i', or 'n'.")

    if profile_has_changed:
        save_profile(profile)

    # Exclude non-Active keys from final Excel outputs
    for item in all_files_data:
        audit_type = item["audit_type"]
        for b in item["blocks"]:
            b["Audit_Source_Type"] = audit_type
            b["Original_File"] = item["file_name"]

            for k in list(b.keys()):
                if k in BYPASS_KEYS:
                    continue

                saved_status = profile.get(audit_type, {}).get(k)
                if saved_status == "Active":
                    b["Key_Classification"] = saved_status
                else:
                    del b[k]


def write_excel_grouped(writer, blocks, sheet_name_prefix):
    """Writes items sequentially to a single sheet with Forest Green styling."""
    df_all = pd.DataFrame(blocks)
    if df_all.empty:
        return

    front_cols = ["Block_Scope", "Condition_Context", "Report_Context", "Audit_Source_Type", "Original_File", "block_type"]
    available_front = [c for c in front_cols if c in df_all.columns]
    other_cols = sorted([c for c in df_all.columns if c not in front_cols])
    df_all = df_all[available_front + other_cols].dropna(how='all', axis=1)

    clean_sheet_name = f"{sheet_name_prefix}_Baselines"[:31]
    df_all.to_excel(writer, sheet_name=clean_sheet_name, index=False)
    
    ws = writer.sheets[clean_sheet_name]
    ws.sheet_properties.tabColor = "27AE60"


def audit_to_excel(input_path: Path, output_path: Path):
    text = input_path.read_text(encoding="utf-8", errors="replace")
    vars_map = parse_variables(text)
    blocks = parse_custom_items(text)
    audit_type = detect_audit_type(text, filename=input_path.name)

    if not blocks:
        return None

    process_tokens(blocks, vars_map)
    file_data = [{"file_name": input_path.name, "blocks": blocks, "vars": vars_map, "audit_type": audit_type}]
    validate_and_classify_keys(file_data)

    with pd.ExcelWriter(output_path, engine="openpyxl") as writer:
        write_excel_grouped(writer, blocks, audit_type)
    return output_path


def audit_dir_to_combined_excel(input_dir: Path, output_path: Path):
    files = sorted(input_dir.glob("*.audit"))
    if not files:
        print(f"WARNING: Zero .audit files discovered inside {input_dir}")
        return output_path
    
    output_path.parent.mkdir(parents=True, exist_ok=True)
    all_files_data = []
    
    for audit_file in files:
        text = audit_file.read_text(encoding="utf-8", errors="replace")
        vars_map = parse_variables(text)
        blocks = parse_custom_items(text)
        audit_type = detect_audit_type(text, filename=audit_file.name)
        
        if blocks:
            process_tokens(blocks, vars_map)
            all_files_data.append({"file_name": audit_file.name, "blocks": blocks, "vars": vars_map, "audit_type": audit_type})
        else:
            print(f" -> Skipping '{audit_file.name}': No configuration blocks extracted.")

    if not all_files_data:
        print("No valid baseline configurations found inside any audit files.")
        return output_path

    validate_and_classify_keys(all_files_data)

    grouped_data = {}
    for item in all_files_data:
        grouped_data.setdefault(item["audit_type"], []).extend(item["blocks"])

    with pd.ExcelWriter(output_path, engine="openpyxl") as writer:
        for audit_type, aggregated_blocks in grouped_data.items():
            if not aggregated_blocks:
                continue
            write_excel_grouped(writer, aggregated_blocks, audit_type)

    return output_path
